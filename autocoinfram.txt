-- Lưu script vào workspace
local function SaveScriptToWorkspace()
    pcall(function()
        local execFolder = workspace:FindFirstChild("AutoExecFolder")
        if not execFolder then
            execFolder = Instance.new("Folder")
            execFolder.Name = "AutoExecFolder"
            execFolder.Parent = workspace
        end
        
        local scriptValue = execFolder:FindFirstChild("AutoFarmScript")
        if not scriptValue then
            scriptValue = Instance.new("StringValue")
            scriptValue.Name = "AutoFarmScript"
            scriptValue.Parent = execFolder
        end
        
        scriptValue.Value = SCRIPT_SOURCE
        print("[Auto Farm] Script saved to workspace")
    end)
end

-- Auto Rejoin
pcall(function()
    repeat task.wait() until game.CoreGui:FindFirstChild('RobloxPromptGui')
    local po = game.CoreGui.RobloxPromptGui.promptOverlay
    local ts = game:GetService('TeleportService')
    
    po.ChildAdded:Connect(function(a)
        if a.Name == 'ErrorPrompt' then
            -- Lưu script trước khi rejoin
            SaveScriptToWorkspace()
            
            while true do
                ts:Teleport(game.PlaceId)
                task.wait(2)
            end
        end
    end)
end)

-- Kiểm tra và execute script từ workspace
local function CheckAndExecute()
    pcall(function()
        local execFolder = workspace:FindFirstChild("AutoExecFolder")
        if execFolder then
            local scriptValue = execFolder:FindFirstChild("AutoFarmScript")
            if scriptValue and scriptValue.Value ~= "" then
                print("[Auto Farm] Executing saved script...")
                loadstring(scriptValue.Value)()
            end
        end
    end)
end
